**Goal:** Train the existing MiniMind checkpoint with assistant-only supervised fine-tuning on a frozen, split-audited chat corpus, then evaluate it against a no-SFT baseline and a mismatched-answer control on 27 held-out questions.

// ============================================================
// chatSftMetrics — Phase 113 answer comparison
// ============================================================
// Deterministic, string-only metrics. No model, no random, no I/O.
// Used by the dataset audit, the evaluation probe, and the tests so
// all three describe the same EM and non-EM comparisons.
// ============================================================

export type ChatSftLanguage = "en" | "zh" | "mixed";

export interface NormalizedAnswer {
  text: string;
  /** true when the raw model output tried to start another turn. */
  turnLeak: boolean;
}

const TRAILING_PUNCTUATION = /[。．.!！?？]+$/u;
const TURN_MARKERS = ["\nUser:", "\nAssistant:"] as const;

/**
 * Frozen answer-normalization contract for the primary exact-match metric.
 *
 * 1. Normalize Unicode to NFC.
 * 2. Normalize CRLF and CR line endings to LF.
 * 3. Trim leading and trailing whitespace.
 * 4. Collapse each run of consecutive whitespace characters to one ASCII space.
 * 5. Preserve case, punctuation, wording, numbers, and semantic content.
 *
 * No fuzzy matching, stemming, semantic similarity, or post-hoc exceptions.
 */
export function normalizeAnswer(text: string): string {
  return text
    .normalize("NFC")
    .replace(/\r\n?/g, "\n")
    .trim()
    .replace(/\s+/g, " ");
}

/**
 * Extract the assistant continuation before normalization. A model output that
 * starts another turn is cut at the first "\nUser:" or "\nAssistant:" marker,
 * and the cut is recorded for the integrity report. This extraction is not part
 * of `normalizeAnswer` and does not change the normalization rule.
 */
export function normalizeChatAnswer(raw: string): NormalizedAnswer {
  let text = raw.replace(/\r\n?/g, "\n");
  let turnLeak = false;

  for (const marker of TURN_MARKERS) {
    const index = text.indexOf(marker);
    if (index !== -1) {
      text = text.slice(0, index);
      turnLeak = true;
    }
  }

  return { text: normalizeAnswer(text), turnLeak };
}

/**
 * Non-EM comparison key used only for duplicate detection and the control
 * mapping's "answers must differ" test. It is deliberately more permissive
 * than `normalizeAnswer`; it never scores a prediction.
 */
export function chatAnswerComparisonKey(raw: string): string {
  return normalizeChatAnswer(raw)
    .text.toLowerCase()
    .replace(TRAILING_PUNCTUATION, "")
    .trim();
}

function splitAnswerTokens(text: string, language: ChatSftLanguage): string[] {
  if (text.length === 0) return [];
  if (language === "en") {
    return text.split(" ").filter((token) => token.length > 0);
  }
  return Array.from(text.replace(/\s+/g, ""));
}

function multiset(tokens: string[]): Map<string, number> {
  const counts = new Map<string, number>();
  for (const token of tokens) {
    counts.set(token, (counts.get(token) ?? 0) + 1);
  }
  return counts;
}

/**
 * Multiset token F1 between a normalized prediction and a normalized gold
 * answer. English is whitespace-tokenized; Chinese/mixed is character-tokenized.
 */
export function chatAnswerTokenF1(
  prediction: string,
  gold: string,
  language: ChatSftLanguage
): number {
  const predictionTokens = splitAnswerTokens(prediction, language);
  const goldTokens = splitAnswerTokens(gold, language);
  if (predictionTokens.length === 0 || goldTokens.length === 0) return 0;

  const predictionCounts = multiset(predictionTokens);
  const goldCounts = multiset(goldTokens);
  let overlap = 0;
  for (const [token, count] of predictionCounts) {
    overlap += Math.min(count, goldCounts.get(token) ?? 0);
  }

  const precision = overlap / predictionTokens.length;
  const recall = overlap / goldTokens.length;
  if (precision + recall === 0) return 0;
  return (2 * precision * recall) / (precision + recall);
}

// ChatSftDataset — Phase 113 assistant-only SFT dataset
// ============================================================
// This module is intentionally separate from TextDataset and from
// datasetSchema.ts. It parses the new chat-sft.json envelope, builds
// chat-formatted samples, and masks loss to the final assistant answer
// plus the terminating <eos>.
//
// The mask is aligned to the PRODUCTION Trainer contract:
//   Trainer passes unshifted inputIds and shifts internally, so
//   mask[s] = 1 means "the prediction at position s of token s+1
//   contributes". There is no second target shift.
// ============================================================

import { Dataset } from "./Dataset";

  | "definition"
  | "arithmetic"
  | "translation"

// ============================================================
// Template
// ============================================================

function turnParts(sample: ChatSftSampleData): string[] {
  const parts: string[] = [];
  for (const turn of sample.turns) {
    parts.push(`${TEMPLATE.user}${turn.user}`);
    parts.push(`${TEMPLATE.assistant}${turn.assistant}`);
  }
  return parts;
}

export function formatFullText(sample: ChatSftSampleData): string {
  return turnParts(sample).join(TEMPLATE.separator);
}

export function formatPrompt(sample: ChatSftSampleData): string {
  const parts = turnParts(sample);
  return [...parts.slice(0, -1), TEMPLATE.assistant.trimEnd()].join(
    TEMPLATE.separator
  );
}

export function formatAnswerContinuation(sample: ChatSftSampleData): string {
  const lastTurn = sample.turns[sample.turns.length - 1];
  return ` ${lastTurn.assistant}`;
}

export function replaceFinalAnswer(
  sample: ChatSftSampleData,
  finalAnswer: string
): ChatSftSampleData {
  const turns = sample.turns.map((turn) => ({ ...turn }));
  turns[turns.length - 1].assistant = finalAnswer;
  return { ...sample, turns };
}

function arraysEqual(a: number[], b: number[]): boolean {
  if (a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) {
    if (a[i] !== b[i]) return false;
  }
  return true;
}

/**
 * Build one padded TextSample from a chat sample.
 *
 * Normative sequence:
 *   encode(promptText) ++ encode(" " + finalAnswer) ++ [<eos>]
 *
 * Segment-wise encoding guarantees the training prefix is exactly the
 * token sequence the inference path sees. Whether the tokenizer would
 * have produced the same sequence from the literal full text is recorded
 * as a diagnostic, never used to reject a sample.
 */
export function encodeChatSftSample(
  sample: ChatSftSampleData,
  tokenizer: BPETokenizer,
  blockSize: number = CHAT_SFT_BLOCK_SIZE
): ChatSftEncoding {
  if (sample.turns.length === 0) {
    throw new Error(`ChatSftDataset: sample ${sample.id} has no turns`);
  }

  const promptText = formatPrompt(sample);
  const continuation = formatAnswerContinuation(sample);
  const fullText = formatFullText(sample);

  const promptIds = tokenizer.encode(promptText);
  const answerIds = tokenizer.encode(continuation);
  const fullIds = tokenizer.encode(fullText);
  const expected = [...promptIds, ...answerIds];

  if (promptIds.length === 0 || answerIds.length === 0) {
    throw new Error(`ChatSftDataset: sample ${sample.id} encoded to an empty segment`);
  }

  const ids = [...expected, CHAT_SFT_EOS_TOKEN_ID];
  if (ids.length > blockSize) {
    throw new Error(
      `ChatSftDataset: sample ${sample.id} needs ${ids.length} tokens, ` +
      `block size is ${blockSize}`
    );
  }

  const inputIds = new Array<number>(blockSize).fill(CHAT_SFT_PAD_TOKEN_ID);
  const targetIds = new Array<number>(blockSize).fill(CHAT_SFT_PAD_TOKEN_ID);
  const mask = new Array<number>(blockSize).fill(0);

  for (let i = 0; i < ids.length; i++) inputIds[i] = ids[i];
  for (let s = 0; s < blockSize - 1; s++) targetIds[s] = inputIds[s + 1];

  const eosIndex = ids.length - 1;
  // mask[s] = 1 means the prediction at s targets ids[s+1].
  // The first answer token is at P and is predicted at P-1.
  // EOS is at N-1 and is predicted at N-2. The EOS position itself
  // predicts padding and is never supervised.
  for (let s = promptIds.length - 1; s <= eosIndex - 1; s++) {
    mask[s] = 1;
  }
  // Position blockSize-1 has no target; the loss never reads it.
  mask[blockSize - 1] = 0;

  const supervisedCount = mask.reduce((sum, value) => sum + value, 0);

  return {
    inputIds,
    targetIds,
    mask,
    promptTokenCount: promptIds.length,
    answerTokenCount: answerIds.length,
    eosIndex,
    supervisedCount,
    fullTextTokenizationMatches: arraysEqual(fullIds, expected),
  };
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function readString(
  object: Record<string, unknown>,
  key: string,
  path: string,
  issues: string[]
): string {
  const value = object[key];
  if (typeof value !== "string") {
    issues.push(`${path}.${key} must be a string`);
    return "";
  }
  return value;
}

function readNumber(
  object: Record<string, unknown>,
  key: string,
  path: string,
  issues: string[]
): number {
  const value = object[key];
  if (typeof value !== "number" || !Number.isFinite(value)) {
    issues.push(`${path}.${key} must be a finite number`);
    return 0;
  }
  return value;
}

function isLanguage(value: string): value is ChatSftLanguage {
  return LANGUAGES.includes(value as ChatSftLanguage);
}

function isSplit(value: string): value is ChatSftSplit {
  return SPLITS.includes(value as ChatSftSplit);
}

function isCategory(value: string): value is ChatSftCategory {
  return CATEGORIES.includes(value as ChatSftCategory);
}

const HAN = /[\u3400-\u4dbf\u4e00-\u9fff\uf900-\ufaff]/;
const LATIN = /[A-Za-z]/;
const ID_PATTERN = /^sft-(en|zh|mixed)-(train|validation|test)-\d{3}$/;
const FORBIDDEN_CONTENT = [
  "\n",
  "\r",
  "User:",
  "Assistant:",
  "<pad>",
  "<unk>",
  "<bos>",
  "<eos>",
];

function contentIssue(text: string): string | null {
  if (text.length === 0) return "must not be empty";
  if (text !== text.trim()) return "must not have leading or trailing whitespace";
  for (const token of FORBIDDEN_CONTENT) {
    if (text.includes(token)) return `must not contain ${JSON.stringify(token)}`;
  }
  return null;
}

function languageMatches(language: ChatSftLanguage, userText: string): boolean {
  const hasHan = HAN.test(userText);
  const hasLatin = LATIN.test(userText);
  switch (language) {
    case "en":
      return !hasHan;
    case "zh":
      return hasHan && !hasLatin;
    case "mixed":
      return hasHan && hasLatin;
  }
}

function parseSample(
  raw: unknown,
  index: number,
  issues: string[]
): ChatSftSampleData | null {
  const path = `$.samples[${index}]`;
  const issueCountBefore = issues.length;
  if (!isRecord(raw)) {
    issues.push(`${path} must be an object`);
    return null;
  }

  const id = readString(raw, "id", path, issues);
  const languageText = readString(raw, "language", path, issues);
  const splitText = readString(raw, "split", path, issues);
  const categoryText = readString(raw, "category", path, issues);

  if (!ID_PATTERN.test(id)) issues.push(`${path}.id has an invalid format`);
  if (!isLanguage(languageText)) issues.push(`${path}.language is invalid`);
  if (!isSplit(splitText)) issues.push(`${path}.split is invalid`);
  if (!isCategory(categoryText)) issues.push(`${path}.category is invalid`);

  const turnsRaw = raw["turns"];
  const turns: ChatSftTurn[] = [];
  if (!Array.isArray(turnsRaw) || (turnsRaw.length !== 1 && turnsRaw.length !== 2)) {
    issues.push(`${path}.turns must contain 1 or 2 turns`);
  } else {
    turnsRaw.forEach((turnRaw, turnIndex) => {
      const turnPath = `${path}.turns[${turnIndex}]`;
      if (!isRecord(turnRaw)) {
        issues.push(`${turnPath} must be an object`);
        return;
      }
      const user = readString(turnRaw, "user", turnPath, issues);
      const assistant = readString(turnRaw, "assistant", turnPath, issues);
      const userIssue = contentIssue(user);
      const assistantIssue = contentIssue(assistant);
      if (userIssue) issues.push(`${turnPath}.user ${userIssue}`);
      if (assistantIssue) issues.push(`${turnPath}.assistant ${assistantIssue}`);
      if (isLanguage(languageText) && !languageMatches(languageText, user)) {
        issues.push(`${turnPath}.user does not match language "${languageText}"`);
      }
      turns.push({ user, assistant });
    });
  }

  if (isLanguage(languageText) && isSplit(splitText) && ID_PATTERN.test(id)) {
    const expectedPrefix = `sft-${languageText}-${splitText}-`;
    if (!id.startsWith(expectedPrefix)) {
      issues.push(`${path}.id does not agree with language/split`);
    }
  }

  if (
    issues.length > issueCountBefore ||
    !isLanguage(languageText) ||
    !isSplit(splitText) ||
    !isCategory(categoryText) ||
    !ID_PATTERN.test(id) ||
    turns.length === 0
  ) {
    return null;
  }

  return {
    id,
    language: languageText,
    split: splitText,
    category: categoryText,
    turns,
  };
}

/**
 * Parse and structurally validate a chat-sft dataset file.
 * @throws with every structural issue, joined by newlines.
 */
export function parseChatSftDatasetFile(value: unknown): ChatSftDatasetFile {
  const issues: string[] = [];
  if (!isRecord(value)) {
    throw new Error("chat-sft dataset must be a JSON object");
  }

  const dataset = readString(value, "dataset", "$", issues);
  const version = readString(value, "version", "$", issues);
  const schemaVersion = readNumber(value, "schemaVersion", "$", issues);
  const purpose = readString(value, "purpose", "$", issues);
  const createdAt = readString(value, "createdAt", "$", issues);

  if (dataset !== "chat-sft") issues.push('$.dataset must be "chat-sft"');
  if (schemaVersion !== 1) issues.push("$.schemaVersion must be 1");
  if (purpose !== "instruction") issues.push('$.purpose must be "instruction"');

  const templateRaw = value["template"];
  let template: ChatSftTemplate = { user: "", assistant: "", separator: "" };
  if (!isRecord(templateRaw)) {
    issues.push("$.template must be an object");
  } else {
    template = {
      user: readString(templateRaw, "user", "$.template", issues),
      assistant: readString(templateRaw, "assistant", "$.template", issues),
      separator: readString(templateRaw, "separator", "$.template", issues),
    };
    if (
      template.user !== TEMPLATE.user ||
      template.assistant !== TEMPLATE.assistant ||
      template.separator !== TEMPLATE.separator
    ) {
      issues.push("$.template does not match the frozen chat template");
    }
  }

  const samplesRaw = value["samples"];
  const samples: ChatSftSampleData[] = [];
  if (!Array.isArray(samplesRaw)) {
    issues.push("$.samples must be an array");
  } else {
    samplesRaw.forEach((raw, index) => {
      const sample = parseSample(raw, index, issues);
      if (sample) samples.push(sample);
    });
  }

  const ids = new Set<string>();
  for (const sample of samples) {
    if (ids.has(sample.id)) issues.push(`duplicate sample id "${sample.id}"`);
    ids.add(sample.id);
  }

  if (issues.length > 0) {
    throw new Error(`Invalid chat-sft dataset:\n- ${issues.join("\n- ")}`);
  }

  return { dataset, version, schemaVersion, purpose, template, createdAt, samples };
}

export class ChatSftDataset extends Dataset<TextSample> {
  private readonly encoded: ChatSftEncoding[];
  private readonly blockSize: number;
  private readonly split: ChatSftSplit;

  constructor(options: {
    tokenizer: BPETokenizer;
    dataset: ChatSftDatasetFile;
    split: ChatSftSplit;
    blockSize?: number;
  }) {
    super();
    const blockSize = options.blockSize ?? CHAT_SFT_BLOCK_SIZE;
    const samples = options.dataset.samples.filter(
      (sample) => sample.split === options.split
    );
    if (samples.length === 0) {
      throw new Error(`ChatSftDataset: split "${options.split}" is empty`);
    }
    this.blockSize = blockSize;
    this.split = options.split;
    this.encoded = samples.map((sample) =>
      encodeChatSftSample(sample, options.tokenizer, blockSize)
    );
  }

  length(): number {
    return this.encoded.length;
  }

  get(index: number): TextSample {
    if (index < 0 || index >= this.encoded.length) {
      throw new Error(
        `ChatSftDataset.get: index ${index} out of range ` +
        `[0, ${this.encoded.length}) for split "${this.split}"`
      );
    }
    const sample = this.encoded[index];
    return {
      inputIds: [...sample.inputIds],
      targetIds: [...sample.targetIds],
      mask: [...sample.mask],
    };
  }

  getBlockSize(): number {
    return this.blockSize;
  }
}

/**
 * Audit split/language counts, turn mix, category quotas, leakage,
 * tokenization boundaries, and held-out prompt length.
 */
export function auditChatSftDataset(
  dataset: ChatSftDatasetFile,
  tokenizer: BPETokenizer
): ChatSftAudit {
  const issues: string[] = [];

  const bySplit: Record<ChatSftSplit, number> = { train: 0, validation: 0, test: 0 };
  const byLanguage: Record<ChatSftLanguage, number> = { en: 0, zh: 0, mixed: 0 };
  const byCategory: Record<ChatSftCategory, number> = {
    factual: 0,
    definition: 0,
    arithmetic: 0,
    translation: 0,
    daily: 0,
    technology: 0,
  };
  const bySplitLanguage: Record<string, number> = {};
  const turnCounts = new Map<string, { single: number; two: number }>();

  for (const sample of dataset.samples) {
    bySplit[sample.split] += 1;
    byLanguage[sample.language] += 1;
    byCategory[sample.category] += 1;

    const splitLanguageKey = `${sample.split}/${sample.language}`;
    bySplitLanguage[splitLanguageKey] = (bySplitLanguage[splitLanguageKey] ?? 0) + 1;

    const turnKey = `${sample.split}/${sample.language}`;
    const counts = turnCounts.get(turnKey) ?? { single: 0, two: 0 };
    if (sample.turns.length === 1) counts.single += 1;
    else counts.two += 1;
    turnCounts.set(turnKey, counts);
  }

  for (const split of SPLITS) {
    for (const language of LANGUAGES) {
      const actual = bySplitLanguage[`${split}/${language}`] ?? 0;
      const expected = EXPECTED_SPLIT_LANGUAGE[split][language];
      if (actual !== expected) {
        issues.push(
          `${split}/${language}: expected ${expected} samples, found ${actual}`
        );
      }

      const counts = turnCounts.get(`${split}/${language}`) ?? { single: 0, two: 0 };
      const expectedTurns = EXPECTED_TURNS[split][language];
      if (counts.single !== expectedTurns.single) {
        issues.push(
          `${split}/${language}: expected ${expectedTurns.single} single-turn, ` +
          `found ${counts.single}`
        );
      }
      if (counts.two !== expectedTurns.two) {
        issues.push(
          `${split}/${language}: expected ${expectedTurns.two} two-turn, ` +
          `found ${counts.two}`
        );
      }

      for (const category of CATEGORIES) {
        const categoryCount = dataset.samples.filter(
          (sample) =>
            sample.split === split &&
            sample.language === language &&
            sample.category === category
        ).length;
        const expectedCategory = EXPECTED_CATEGORY[split][category];
        if (categoryCount !== expectedCategory) {
          issues.push(
            `${split}/${language}/${category}: expected ${expectedCategory}, ` +
            `found ${categoryCount}`
          );
        }
      }
    }
  }

  const trainingValidation = dataset.samples.filter(
    (sample) => sample.split !== "test"
  );
  const heldOut = dataset.samples.filter((sample) => sample.split === "test");

  const seenQuestions = new Map<string, string>();
  const seenAnswers = new Map<string, string>();
  for (const sample of trainingValidation) {
    for (const turn of sample.turns) {
      const question = chatAnswerComparisonKey(turn.user);
      const answer = chatAnswerComparisonKey(turn.assistant);
      const questionOwner = seenQuestions.get(question);
      if (questionOwner !== undefined) {
        issues.push(`duplicate training question in ${questionOwner} and ${sample.id}`);
      } else {
        seenQuestions.set(question, sample.id);
      }
      const answerOwner = seenAnswers.get(answer);
      if (answerOwner !== undefined) {
        issues.push(`duplicate training answer in ${answerOwner} and ${sample.id}`);
      } else {
        seenAnswers.set(answer, sample.id);
      }
    }
  }

  const trainingPrompts = new Set(
    trainingValidation.map((sample) => chatAnswerComparisonKey(formatPrompt(sample)))
  );

  for (const sample of heldOut) {
    for (const turn of sample.turns) {
      const question = chatAnswerComparisonKey(turn.user);
      const answer = chatAnswerComparisonKey(turn.assistant);
      if (seenQuestions.has(question)) {
        issues.push(`held-out question of ${sample.id} occurs in training`);
      }
      if (seenAnswers.has(answer)) {
        issues.push(`held-out answer of ${sample.id} occurs in training`);
      }
    }
    if (trainingPrompts.has(chatAnswerComparisonKey(formatPrompt(sample)))) {
      issues.push(`held-out prompt of ${sample.id} occurs in training`);
    }
  }

  let exact = 0;
  let segmented = 0;
  let maxTestPromptTokens = 0;
  const boundaryDetails: Array<{ id: string; matches: boolean }> = [];

  for (const sample of dataset.samples) {
    const encoded = encodeChatSftSample(sample, tokenizer);
    if (encoded.fullTextTokenizationMatches) exact += 1;
    else segmented += 1;
    boundaryDetails.push({
      id: sample.id,
      matches: encoded.fullTextTokenizationMatches,
    });
    if (sample.split === "test") {
      maxTestPromptTokens = Math.max(maxTestPromptTokens, encoded.promptTokenCount);
      if (encoded.promptTokenCount > 88) {
        issues.push(
          `${sample.id}: prompt is ${encoded.promptTokenCount} tokens ` +
          `(maximum 88 so prompt + 40 never truncates context)`
        );
      }
    }
  }

  const singleTurn = dataset.samples.filter((sample) => sample.turns.length === 1).length;
  const twoTurn = dataset.samples.length - singleTurn;

  return {
    total: dataset.samples.length,
    bySplit,
    bySplitLanguage,
    byLanguage,
    byCategory,
    singleTurn,
    twoTurn,
    boundary: { exact, segmented },
    boundaryDetails,
    maxTestPromptTokens,
    issues,
  };
}

export interface ChatSftControlEntry {
  sampleId: string;
  sourceSampleId: string;
  offset: number;
  finalAnswer: string;
}

export interface ChatSftControlStratum {
  split: "train" | "validation";
  language: ChatSftLanguage;
  turnCount: number;
  size: number;
  offset: number;
}

export interface ChatSftControlMapping {
  datasetSha256: string;
  tokenizerFingerprint: string;
  strata: ChatSftControlStratum[];
  entries: ChatSftControlEntry[];
}

/**
 * Build the deterministic mismatched-answer mapping for train/validation.
 *
 * Within each (split, language, turnCount) stratum, samples are sorted by
 * id and final answers are cyclically shifted by the smallest offset that
 * (a) never maps an answer to itself and (b) fits the block size. Only the
 * final supervised answer changes; prompts stay byte-identical to B.
 */
export function buildChatSftControlMapping(
  dataset: ChatSftDatasetFile,
  tokenizer: BPETokenizer,
  datasetSha256: string,
  tokenizerFingerprint: string,
  blockSize: number = CHAT_SFT_BLOCK_SIZE
): ChatSftControlMapping {
  const strata: ChatSftControlStratum[] = [];
  const entries: ChatSftControlEntry[] = [];

  for (const split of ["train", "validation"] as const) {
    for (const language of LANGUAGES) {
      for (const turnCount of [1, 2]) {
        const group = dataset.samples
          .filter(
            (sample) =>
              sample.split === split &&
              sample.language === language &&
              sample.turns.length === turnCount
          )
          .sort((a, b) => a.id.localeCompare(b.id));

        if (group.length === 0) continue;
        if (group.length === 1) {
          throw new Error(
            `Control mapping: stratum ${split}/${language}/${turnCount} has one sample`
          );
        }

        let chosen = -1;
        for (let offset = 1; offset < group.length; offset++) {
          const valid = group.every((target, index) => {
            const source = group[(index + offset) % group.length];
            const sourceAnswer = source.turns[source.turns.length - 1].assistant;
            const targetAnswer = target.turns[target.turns.length - 1].assistant;
            if (
              chatAnswerComparisonKey(sourceAnswer) ===
              chatAnswerComparisonKey(targetAnswer)
            ) {
              return false;
            }
            try {
              encodeChatSftSample(
                replaceFinalAnswer(target, sourceAnswer),
                tokenizer,
                blockSize
              );
              return true;
            } catch {
              return false;
            }
          });
          if (valid) {
            chosen = offset;
            break;
          }
        }

        if (chosen === -1) {
          throw new Error(
            `Control mapping: no valid offset for ${split}/${language}/${turnCount}`
          );
        }

        strata.push({
          split,
          language,
          turnCount,
          size: group.length,
          offset: chosen,
        });

        group.forEach((target, index) => {
          const source = group[(index + chosen) % group.length];
          entries.push({
            sampleId: target.id,
            sourceSampleId: source.id,
            offset: chosen,
            finalAnswer: source.turns[source.turns.length - 1].assistant,
          });
        });
      }
    }
  }

  return { datasetSha256, tokenizerFingerprint, strata, entries };
}

/** Apply a control mapping to train/validation; test samples are untouched. */
export function applyChatSftControlMapping(
  dataset: ChatSftDatasetFile,
  mapping: ChatSftControlMapping
): ChatSftDatasetFile {
  const entryById = new Map(mapping.entries.map((entry) => [entry.sampleId, entry]));
  const sourceById = new Map(dataset.samples.map((sample) => [sample.id, sample]));

  return {
    ...dataset,
    samples: dataset.samples.map((sample) => {
      if (sample.split === "test") return sample;
      const entry = entryById.get(sample.id);
      if (!entry) {
        throw new Error(`Control mapping: no entry for ${sample.id}`);
      }
      const source = sourceById.get(entry.sourceSampleId);
      if (!source) {
        throw new Error(`Control mapping: unknown source ${entry.sourceSampleId}`);
      }
      const sourceAnswer = source.turns[source.turns.length - 1].assistant;
      if (sourceAnswer !== entry.finalAnswer) {
        throw new Error(`Control mapping: answer drift for ${sample.id}`);
      }
      return replaceFinalAnswer(sample, entry.finalAnswer);
    }),
  };
}

/** Validate a control-mapping JSON object. */
export function parseChatSftControlMapping(value: unknown): ChatSftControlMapping {
  if (!isRecord(value)) {
    throw new Error("control mapping must be an object");
  }
  const datasetSha256 = value["datasetSha256"];
  const tokenizerFingerprint = value["tokenizerFingerprint"];
  const strataRaw = value["strata"];
  const entriesRaw = value["entries"];

  if (typeof datasetSha256 !== "string") {
    throw new Error("control mapping datasetSha256 must be a string");
  }
  if (typeof tokenizerFingerprint !== "string") {
    throw new Error("control mapping tokenizerFingerprint must be a string");
  }
  if (!Array.isArray(strataRaw) || !Array.isArray(entriesRaw)) {
    throw new Error("control mapping strata/entries must be arrays");
  }

  const strata: ChatSftControlStratum[] = [];
  for (const raw of strataRaw) {
    if (!isRecord(raw)) throw new Error("control mapping stratum must be an object");
    const split = raw["split"];
    const language = raw["language"];
    const turnCount = raw["turnCount"];
    const size = raw["size"];
    const offset = raw["offset"];
    if (split !== "train" && split !== "validation") {
      throw new Error("control mapping stratum split is invalid");
    }
    if (typeof language !== "string" || !isLanguage(language)) {
      throw new Error("control mapping stratum language is invalid");
    }
    if (
      typeof turnCount !== "number" ||
      typeof size !== "number" ||
      typeof offset !== "number"
    ) {
      throw new Error("control mapping stratum counts must be numbers");
    }
    strata.push({ split, language, turnCount, size, offset });
  }

  const entries: ChatSftControlEntry[] = [];
  for (const raw of entriesRaw) {
    if (!isRecord(raw)) throw new Error("control mapping entry must be an object");
    const sampleId = raw["sampleId"];
    const sourceSampleId = raw["sourceSampleId"];
    const offset = raw["offset"];
    const finalAnswer = raw["finalAnswer"];
    if (
      typeof sampleId !== "string" ||
      typeof sourceSampleId !== "string" ||
      typeof offset !== "number" ||
      typeof finalAnswer !== "string"
    ) {
      throw new Error("control mapping entry has invalid fields");
    }
    entries.push({ sampleId, sourceSampleId, offset, finalAnswer });
  }

  return { datasetSha256, tokenizerFingerprint, strata, entries };
}

function loadBaseWeights(model: TensorMiniMindModel): void {
  const raw = fs.readFileSync(BASE_CKPT, "utf8");

    );

  const checkpoint = JSON.parse(raw) as CheckpointData;
  // Throwaway optimizer: loadCheckpoint restores the checkpoint's LR and step
  // counter, which must not leak into the SFT recipe.
  const scratch = new AdamWOptimizer(model.parameters(), { lr: 1e-5 });
  loadCheckpoint({ model, optimizer: scratch, checkpoint });

function tokenWeightedValidationLoss(
  model: TensorMiniMindModel,
  loader: DataLoader<TextSample>
): number {
  let weightedLoss = 0;
  let tokens = 0;

  for (const batch of loader) {
    const sample = batch[0];
    if (!sample || !sample.mask) {
      throw new Error("Validation batch is missing a mask");
    }
    const { inputIds } = collateBatch([sample]);
    const logits = model.forward(inputIds);
    const loss = crossEntropyLoss(logits, [sample.inputIds], [sample.mask]).item();
    const count = sample.mask.reduce((sum, value) => sum + value, 0);
    weightedLoss += loss * count;
    tokens += count;
  }

  if (tokens === 0) return Number.NaN;
  return weightedLoss / tokens;
}

  tokenizerSource: "metadata" | "reconstructed";
}

interface ItemResult {
  id: string;
  language: ChatSftLanguage;
  category: string;
  prompt: string;
  continuation: string;
  normalized: string;
  gold: string;
  correct: boolean;
  tokenF1: number;
  stopReason: "maxTokens" | "eos" | "error";
  tokenCount: number;

  contextTruncated: boolean;
  utf8Enforced: boolean;
  utf8Invalid: boolean;
  droppedIncompleteTail: boolean;
  emptyCandidateStop: boolean;

interface Aggregate {
  /** Integer exact-match count over `total`; the rate is correct / total. */
  correct: number;
  total: number;
  meanF1: number;

}

interface Integrity {
  /** P012 enforcement was active for every item (must be true). */
  utf8Enforced: boolean;
  invalidUtf8: number;
  droppedIncompleteTail: number;
  emptyCandidateStop: number;
  eosStops: number;
  maxTokenStops: number;
  errorStops: number;
  emptyOutputs: number;
  turnLeaks: number;
  contextTruncations: number;
}

function sha256(data: string | Buffer): string {
  return crypto.createHash("sha256").update(data).digest("hex");
}

function writeJsonAtomic(filePath: string, value: unknown): void {
  const tempPath = `${filePath}.tmp`;
  fs.writeFileSync(tempPath, JSON.stringify(value, null, 2), "utf8");
  fs.rmSync(filePath, { force: true });
  fs.renameSync(tempPath, filePath);
}

function buildTokenizer(): BPETokenizer {
  Random.setSeed(42);
  const tokenizer = new BPETokenizer();
  tokenizer.train(getDatasetTexts(loadProductionDataset()), 300);
  return tokenizer;
}

function buildModel(): TensorMiniMindModel {
  const config = getProductionModelConfig();
  return new TensorMiniMindModel({
    vocabSize: config.vocabSize,
    dModel: config.dModel,
    numHeads: config.numHeads,
    dFF: config.dFF,
    numLayers: config.numLayers,
    maxSeqLen: config.maxSeqLen,
  });
}

function loadArm(
  id: ArmId,
  checkpointPath: string,
  corpus: string[],
  expectedTokenizerFingerprint: string
): Arm {
  const raw = fs.readFileSync(checkpointPath, "utf8");
  const analysis = analyzeCheckpointJSON(raw, corpus);
  if (!analysis.ok) {
    throw new Error(`Condition ${id} failed checkpoint validation: ${analysis.error}`);
  }

  const stateFingerprint = sha256(JSON.stringify(analysis.analysis.tokenizerState));
  if (stateFingerprint !== expectedTokenizerFingerprint) {
    throw new Error(
      `Condition ${id} tokenizer fingerprint mismatch: ` +
      `${stateFingerprint} !== ${expectedTokenizerFingerprint}`
    );
  }
  if (id !== "A" && analysis.analysis.tokenizerSource !== "metadata") {
    throw new Error(`Condition ${id} checkpoint lacks metadata.tokenizerState`);
  }

  const model = buildModel();
  // Throwaway optimizer: only weights are needed for generation.
  const scratch = new AdamWOptimizer(model.parameters(), { lr: 1e-5 });
  loadCheckpoint({ model, optimizer: scratch, checkpoint: analysis.analysis.checkpoint });

  return {
    id,
    model,
    checkpointPath,
    checkpointSha256: sha256(raw),
    tokenizerSource: analysis.analysis.tokenizerSource,
  };
}

function continuationOf(prompt: string, result: GenerationEvalResult): string {
  if (result.generatedText.startsWith(prompt)) {
    return result.generatedText.slice(prompt.length);
  }
  return result.generatedText;
}

function evaluateArm(
  arm: Arm,
  tokenizer: BPETokenizer,
  samples: ChatSftSampleData[],
  decoding: Decoding
): ItemResult[] {
  const evaluator = new GenerationEvaluator(arm.model, tokenizer);
  const prompts = samples.map((sample) => formatPrompt(sample));
  const results = evaluator.evaluate(prompts, {
    maxTokens: 40,
    temperature: decoding === "greedy" ? 0 : 0.8,
    topK: 40,
    topP: 0.9,
    seed: 42,
    eosTokenId: 3,
    enforceUtf8: true,
  });

  return results.map((result, index) => {
    const sample = samples[index];
    const gold = normalizeChatAnswer(sample.turns[0].assistant).text;
    const continuation = continuationOf(result.prompt, result);
    const normalized = normalizeChatAnswer(continuation);
    const promptTokens = tokenizer.encode(result.prompt).length;

    return {
      id: sample.id,
      language: sample.language,
      category: sample.category,
      prompt: result.prompt,
      continuation,
      normalized: normalized.text,
      gold,
      correct: normalized.text.length > 0 && normalized.text === gold,
      tokenF1: chatAnswerTokenF1(normalized.text, gold, sample.language),
      stopReason: result.stopReason,
      tokenCount: result.tokenCount,
      turnLeak: normalized.turnLeak,
      contextTruncated: promptTokens + result.generatedIds.length > 128,
      utf8Enforced: result.utf8?.enforced ?? false,
      utf8Invalid: (result.utf8?.invalidByteOffset ?? -1) !== -1,
      droppedIncompleteTail: result.utf8?.droppedIncompleteTail ?? false,
      emptyCandidateStop: result.utf8?.emptyCandidateStop ?? false,
    };
  });
}

function aggregate(items: ItemResult[]): Aggregate {
  const byLanguage: Record<ChatSftLanguage, { correct: number; total: number }> = {
    en: { correct: 0, total: 0 },
    zh: { correct: 0, total: 0 },
    mixed: { correct: 0, total: 0 },
  };
  let correct = 0;
  let f1 = 0;
  for (const item of items) {
    byLanguage[item.language].total += 1;
    if (item.correct) {
      correct += 1;
      byLanguage[item.language].correct += 1;
    }
    f1 += item.tokenF1;
  }
  return {
    correct,
    total: items.length,
    meanF1: items.length > 0 ? f1 / items.length : 0,
    byLanguage,
  };
}

function integrity(items: ItemResult[]): Integrity {
  return {
    utf8Enforced: items.every((item) => item.utf8Enforced),
    invalidUtf8: items.filter((item) => item.utf8Invalid).length,
    droppedIncompleteTail: items.filter((item) => item.droppedIncompleteTail).length,
    emptyCandidateStop: items.filter((item) => item.emptyCandidateStop).length,
    eosStops: items.filter((item) => item.stopReason === "eos").length,
    maxTokenStops: items.filter((item) => item.stopReason === "maxTokens").length,
    errorStops: items.filter((item) => item.stopReason === "error").length,
    emptyOutputs: items.filter((item) => item.normalized.length === 0).length,
    turnLeaks: items.filter((item) => item.turnLeak).length,
    contextTruncations: items.filter((item) => item.contextTruncated).length,
  };
}

function main(): void {
  const rawDataset = fs.readFileSync(DATASET_PATH, "utf8");
  const dataset: ChatSftDatasetFile = parseChatSftDatasetFile(
    JSON.parse(rawDataset) as unknown
  );
  const tokenizer = buildTokenizer();
  const tokenizerFingerprint = sha256(JSON.stringify(tokenizer.toState()));
  const corpus = getDatasetTexts(loadProductionDataset());

  // File order is frozen: the same array order feeds every condition and both
  // decoding settings.
  const testSamples = dataset.samples.filter(
    (sample) => sample.split === "test"
  );
  if (testSamples.length !== 27) {
    throw new Error(`Expected 27 held-out samples, found ${testSamples.length}`);
  }
  for (const language of ["en", "zh", "mixed"] as const) {
    const count = testSamples.filter((sample) => sample.language === language).length;
    if (count !== 9) {
      throw new Error(`Expected 9 ${language} held-out samples, found ${count}`);
    }
  }
  let promptsOver88 = 0;
  for (const sample of testSamples) {
    const promptTokens = tokenizer.encode(formatPrompt(sample)).length;
    if (promptTokens > 88) promptsOver88 += 1;
  }
  if (promptsOver88 > 0) {
    throw new Error(
      `${promptsOver88} held-out prompt(s) exceed 88 tokens; ` +
      `evaluation would truncate context`
    );
  }

  const armPaths: Record<ArmId, string> = {
    A: path.join(ROOT, "experiments", "phase90", "phase90-aligned.ckpt.json"),
    B: path.join(OUT_DIR, "phase113-sft.ckpt.json"),
    C: path.join(OUT_DIR, "phase113-control.ckpt.json"),
  };

  const arms: Arm[] = (["A", "B", "C"] as const).map((id) =>
    loadArm(id, armPaths[id], corpus, tokenizerFingerprint)
  );
  if (sha256(fs.readFileSync(armPaths.A)) !== BASE_SHA256) {
    throw new Error("Baseline checkpoint changed");
  }

  const decodings: Decoding[] = ["greedy", "sampled"];
  const items: Record<string, ItemResult[]> = {};
  const aggregates: Record<string, Aggregate> = {};
  const integrities: Record<string, Integrity> = {};

  for (const arm of arms) {
    for (const decoding of decodings) {
      const key = `${arm.id}/${decoding}`;
      const armItems = evaluateArm(arm, tokenizer, testSamples, decoding);
      items[key] = armItems;
      aggregates[key] = aggregate(armItems);
      integrities[key] = integrity(armItems);
      console.log(
        `${key}: correct ${aggregates[key].correct}/${aggregates[key].total}, ` +
        `meanF1 ${aggregates[key].meanF1.toFixed(4)}`
      );
    }
  }

  const b = aggregates["B/greedy"];
  const a = aggregates["A/greedy"];
  const c = aggregates["C/greedy"];
  // All criteria use integer exact-match counts over the same 27 items.
  // The margin is two items, never a rate difference of 2.
  const criteria = {
    absolute: b.correct >= 5,
    beatsA: b.correct > a.correct,
    beatsC: b.correct > c.correct,
    margin: b.correct - Math.max(a.correct, c.correct) >= 2,
    perLanguage: (["en", "zh", "mixed"] as const).every(
      (language) => b.byLanguage[language].correct >= 1
    ),
  };

  const integrityBlocked =
    !integrities["A/greedy"].utf8Enforced ||
    !integrities["B/greedy"].utf8Enforced ||
    !integrities["C/greedy"].utf8Enforced ||
    integrities["A/greedy"].contextTruncations > 0 ||
    integrities["B/greedy"].contextTruncations > 0 ||
    integrities["C/greedy"].contextTruncations > 0;

  const allCriteria = Object.values(criteria).every(Boolean);
  const grade = integrityBlocked
    ? "D"
    : allCriteria
      ? "A"
      : criteria.beatsA && criteria.beatsC
        ? "B"
        : "C";
  const label =
    grade === "A"
      ? "Supported"
      : grade === "B"
        ? "Partial"
        : grade === "C"
          ? "Not supported"
          : "Blocked / Inconclusive";

  const verdict = {
    phase: "113",
    hypothesis: "H113",
    grade,
    label,
    criteria,
    primary: { A: a, B: b, C: c },
    integrityBlocked,
    evaluatedItems: testSamples.length,
    promptsOver88,
    datasetSha256: sha256(rawDataset),
    tokenizerFingerprint,
    arms: arms.map((arm) => ({
      id: arm.id,
      checkpointPath: path.relative(ROOT, arm.checkpointPath).replace(/\\/g, "/"),
      checkpointSha256: arm.checkpointSha256,
      tokenizerSource: arm.tokenizerSource,
    })),
  };

  writeJsonAtomic(path.join(OUT_DIR, "eval-results.json"), {
    phase: "113",
    items,
    aggregates,
    integrities,
    verdict,
  });
  writeJsonAtomic(path.join(OUT_DIR, "verdict.json"), verdict);
  fs.writeFileSync(REPORT_PATH, renderReport(verdict, aggregates, integrities), "utf8");

  console.log(`verdict: ${grade} — ${label}`);
}

function writeBlockedArtifacts(error: unknown): void {

    phase: "113",

    grade: "D",
    label: "Blocked / Inconclusive",
    error: message,
    timestamp: Date.now(),
  };
  writeJsonAtomic(path.join(OUT_DIR, "verdict.json"), blocked);
  fs.writeFileSync(

    `\`\`\`\n${message}\n\`\`\`\n`,
    "utf8"
  );
  console.error(`BLOCKED: ${message}`);
}

function renderReport(
  verdict: {
    grade: string;
    label: string;
    criteria: Record<string, boolean>;
    arms: Array<{ id: string; checkpointSha256: string; tokenizerSource: string }>;
    datasetSha256: string;
    tokenizerFingerprint: string;
  },
  aggregates: Record<string, Aggregate>,
  integrities: Record<string, Integrity>
): string {
  const rows = (["A", "B", "C"] as const)
    .map((arm) => {
      const greedy = aggregates[`${arm}/greedy`];
      const sampled = aggregates[`${arm}/sampled`];
      return (
        `| ${arm} | ${greedy.correct}/${greedy.total} | ${greedy.meanF1.toFixed(4)} | ` +
        `${sampled.correct}/${sampled.total} | ` +
        `${greedy.byLanguage.en.correct}/${greedy.byLanguage.en.total} | ` +
        `${greedy.byLanguage.zh.correct}/${greedy.byLanguage.zh.total} | ` +
        `${greedy.byLanguage.mixed.correct}/${greedy.byLanguage.mixed.total} |`
      );
    })
    .join("\n");

  const criteriaRows = Object.entries(verdict.criteria)
    .map(([name, passed]) => `| ${name} | ${passed ? "PASS" : "FAIL"} |`)
    .join("\n");

  const integrityRows = (["A", "B", "C"] as const)
    .map((arm) => {
      const value = integrities[`${arm}/greedy`];
      return (
        `| ${arm} | ${value.utf8Enforced ? "yes" : "NO"} | ` +
        `${value.invalidUtf8} | ${value.droppedIncompleteTail} | ` +
        `${value.emptyCandidateStop} | ${value.eosStops} | ${value.maxTokenStops} | ` +
        `${value.errorStops} | ${value.emptyOutputs} | ${value.turnLeaks} | ` +
        `${value.contextTruncations} |`
      );
    })
    .join("\n");

  return `# Phase 113 — Minimal Chat SFT — Report

**Date:** ${new Date().toISOString().slice(0, 10)}
**Verdict:** ${verdict.grade} — ${verdict.label}

## 1. Question

Does assistant-only SFT improve held-out chat answers versus no SFT (A) and a
mismatched-answer control (C)?

## 2. Frozen inputs

- Dataset SHA-256: \`${verdict.datasetSha256}\`
- Tokenizer fingerprint: \`${verdict.tokenizerFingerprint}\`

## 3. Primary metric (greedy normalized exact-match count)

| Arm | Greedy EM count | mean token F1 | Sampled EM count | en EM count | zh EM count | mixed EM count |
|---|---|---|---|---|---|---|
${rows}

Counts are integers out of 27; the corresponding rate is \`count / 27\`.

## 4. Acceptance criteria

| Criterion | Result |
|---|---|
${criteriaRows}

All criteria use integer exact-match counts over the same 27 held-out questions
(A=${aggregates["A/greedy"].correct}, B=${aggregates["B/greedy"].correct},
C=${aggregates["C/greedy"].correct}):

- \`absolute\`: \`correct_B >= 5\`
- \`beatsA\`: \`correct_B > correct_A\`
- \`beatsC\`: \`correct_B > correct_C\`
- \`margin\`: \`correct_B - max(correct_A, correct_C) >= 2\`
- \`perLanguage\`: \`correct_B >= 1\` in each of \`en\`, \`zh\`, \`mixed\`

The margin is two items (\`2/27\` as a rate), never a rate difference of 2.
Token F1 is secondary and cannot substitute for these criteria.

## 5. Integrity diagnostics (not capability)

| Arm | UTF-8 enforced | invalid UTF-8 | dropped tail | empty candidate | EOS stops | maxToken stops | error stops | empty outputs | turn leaks | ctx truncations |
|---|---|---|---|---|---|---|---|---|---|---|
${integrityRows}

P012 is a decoding safeguard. A UTF-8-clean answer is not counted as correct
unless it matches the gold answer.

## 6. Artifact hashes

${verdict.arms
  .map((arm) => `- ${arm.id}: \`${arm.checkpointSha256}\` (tokenizer: ${arm.tokenizerSource})`)
  .join("\n")}

## 7. Limitations

- 84 training examples and a 1.1M-parameter model; 27 held-out items.
- The held-out set is single-turn; multi-turn behavior is trained but not claimed.
- The base tokenizer state was reconstructed, not recorded.
- EOS termination is newly learned.
- No causal claim beyond this controlled experiment.
`;
}
